set count pixels cardMemory;

// water images keep their pixels in runtime memory, everything else in physical memory
set blockcond pixels semantic == TS_WATER_MAP XFILE_BLOCK_RUNTIME XFILE_BLOCK_PHYSICAL;
reorder:
    name
    pixels;
